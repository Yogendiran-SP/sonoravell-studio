cmake_minimum_required(VERSION 3.20)
project(SonoravellStudio LANGUAGES CXX)

add_library(sonoravell_engine native/session.cpp native/project.cpp native/take_journal.cpp)
target_include_directories(sonoravell_engine PUBLIC native)
target_compile_features(sonoravell_engine PUBLIC cxx_std_20)

add_executable(sonoravell-render native/render_cli.cpp)
target_link_libraries(sonoravell-render PRIVATE sonoravell_engine)

if(WIN32)
  add_executable(SonoravellStudio WIN32 native/windows.cpp native/windows_capture.cpp)
  target_link_libraries(SonoravellStudio PRIVATE sonoravell_engine winmm comdlg32 gdi32)
  install(TARGETS SonoravellStudio sonoravell-render RUNTIME DESTINATION .)
  install(FILES README.md LICENSE DESTINATION .)
endif()

find_package(Threads REQUIRED)
include(CTest)
if(BUILD_TESTING)
  add_executable(session_check tests/session_check.cpp)
  target_link_libraries(session_check PRIVATE sonoravell_engine)
  add_test(NAME session_check COMMAND session_check)
  add_executable(project_check tests/project_check.cpp)
  target_link_libraries(project_check PRIVATE sonoravell_engine)
  add_test(NAME project_check COMMAND project_check)
  add_executable(take_check tests/take_check.cpp)
  target_link_libraries(take_check PRIVATE sonoravell_engine Threads::Threads)
  add_test(NAME take_check COMMAND take_check)
  if(WIN32)
    add_test(NAME workspace_check COMMAND SonoravellStudio --workspace-check)
  endif()
endif()
