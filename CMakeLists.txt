cmake_minimum_required(VERSION 3.20)
project(SonoravellStudio LANGUAGES CXX)

add_library(sonoravell_engine native/session.cpp native/project.cpp native/take_journal.cpp native/session_autosave.cpp
  native/shared_render.cpp native/shared_capture.cpp native/stream_clock.cpp native/duplex_coordination.cpp
  native/duplex_transport.cpp native/loopback_qualification.cpp native/duplex_qualification.cpp)
target_include_directories(sonoravell_engine PUBLIC native)
target_compile_features(sonoravell_engine PUBLIC cxx_std_20)

add_executable(sonoravell-render native/render_cli.cpp)
target_link_libraries(sonoravell-render PRIVATE sonoravell_engine)
add_executable(sonoravell-qualify native/qualify_cli.cpp)
target_link_libraries(sonoravell-qualify PRIVATE sonoravell_engine)

if(WIN32)
  add_executable(SonoravellStudio WIN32 native/windows.cpp native/windows_capture.cpp native/windows_devices.cpp
    native/windows_wasapi.cpp native/windows_wasapi_capture.cpp native/windows_wasapi_endpoints.cpp)
  target_link_libraries(SonoravellStudio PRIVATE sonoravell_engine winmm comdlg32 gdi32 ole32)
  install(TARGETS SonoravellStudio sonoravell-render sonoravell-qualify RUNTIME DESTINATION .)
  install(FILES README.md LICENSE DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
endif()

find_package(Threads REQUIRED)
include(CTest)
if(BUILD_TESTING)
  add_executable(session_check tests/session_check.cpp)
  target_link_libraries(session_check PRIVATE sonoravell_engine)
  add_test(NAME session_check COMMAND session_check)
  add_executable(project_check tests/project_check.cpp)
  target_link_libraries(project_check PRIVATE sonoravell_engine)
  add_test(NAME project_check COMMAND project_check)
  add_executable(take_check tests/take_check.cpp)
  target_link_libraries(take_check PRIVATE sonoravell_engine Threads::Threads)
  add_test(NAME take_check COMMAND take_check)
  add_executable(shared_render_check tests/shared_render_check.cpp)
  target_link_libraries(shared_render_check PRIVATE sonoravell_engine)
  add_test(NAME shared_render_check COMMAND shared_render_check)
  add_executable(shared_capture_check tests/shared_capture_check.cpp)
  target_link_libraries(shared_capture_check PRIVATE sonoravell_engine)
  add_test(NAME shared_capture_check COMMAND shared_capture_check)
  add_executable(stream_clock_check tests/stream_clock_check.cpp)
  target_link_libraries(stream_clock_check PRIVATE sonoravell_engine)
  add_test(NAME stream_clock_check COMMAND stream_clock_check)
  add_executable(duplex_coordination_check tests/duplex_coordination_check.cpp)
  target_link_libraries(duplex_coordination_check PRIVATE sonoravell_engine)
  add_test(NAME duplex_coordination_check COMMAND duplex_coordination_check)
  add_executable(duplex_transport_check tests/duplex_transport_check.cpp)
  target_link_libraries(duplex_transport_check PRIVATE sonoravell_engine Threads::Threads)
  add_test(NAME duplex_transport_check COMMAND duplex_transport_check)

  add_executable(duplex_qualification_check tests/duplex_qualification_check.cpp)
  target_link_libraries(duplex_qualification_check PRIVATE sonoravell_engine)
